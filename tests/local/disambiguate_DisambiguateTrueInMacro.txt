>>===== MODE =====>>
citation
<<===== MODE =====<<


>>===== DESCRIPTION =====>>
A citation calls a macro containing <if disambiguate="true">. The condition
must be indexed and expanded even though it is not directly in the citation
layout.
<<===== DESCRIPTION =====<<


>>===== RESULT =====>>
Smith Book A 1980; Smith Book B 1980
<<===== RESULT =====<<


>>===== CSL =====>>
<style
      xmlns="http://purl.org/net/xbiblio/csl"
      class="in-text"
      version="1.0">
  <info>
    <id />
    <title />
    <updated>2026-11-04T00:00:00+00:00</updated>
  </info>
  <macro name="citation-author-year">
    <group delimiter=" ">
      <names variable="author">
        <name form="short"/>
      </names>
      <choose>
        <if disambiguate="true">
          <text variable="title"/>
        </if>
      </choose>
      <date variable="issued">
        <date-part name="year"/>
      </date>
    </group>
  </macro>
  <citation>
    <layout delimiter="; ">
      <text macro="citation-author-year"/>
    </layout>
  </citation>
</style>
<<===== CSL =====<<
